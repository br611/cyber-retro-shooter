    <script>
        const ThemeConfig = {
            title: "THREAT NEUTRALIZER",
            tagline: "Destroy cybersecurity vulnerabilities instantly.",
            playerColor: "#00f3ff",
            enemyColor: "#ff007f",
            laserColor: "#00ff66",
            starSpeed: 2
        };

        const canvas = document.getElementById('gameCanvas');
        const ctx = canvas.getContext('2d');

        const SoundEngine = {
            ctx: null,
            init() {
                this.ctx = new (window.AudioContext || window.webkitAudioContext)();
            },
            playLaser() {
                if (!this.ctx) return;
                let osc = this.ctx.createOscillator();
                let gain = this.ctx.createGain();
                osc.type = 'sawtooth';
                osc.frequency.setValueAtTime(800, this.ctx.currentTime);
                osc.frequency.exponentialRampToValueAtTime(150, this.ctx.currentTime + 0.15);
                gain.gain.setValueAtTime(0.1, this.ctx.currentTime);
                gain.gain.linearRampToValueAtTime(0.001, this.ctx.currentTime + 0.15);
                osc.connect(gain);
                gain.connect(this.ctx.destination);
                osc.start();
                osc.stop(this.ctx.currentTime + 0.15);
            },
            playExplosion() {
                if (!this.ctx) return;
                let osc = this.ctx.createOscillator();
                let gain = this.ctx.createGain();
                osc.type = 'triangle';
                osc.frequency.setValueAtTime(160, this.ctx.currentTime);
                osc.frequency.linearRampToValueAtTime(40, this.ctx.currentTime + 0.3);
                gain.gain.setValueAtTime(0.2, this.ctx.currentTime);
                gain.gain.exponentialRampToValueAtTime(0.001, this.ctx.currentTime + 0.3);
                osc.connect(gain);
                gain.connect(this.ctx.destination);
                osc.start();
                osc.stop(this.ctx.currentTime + 0.3);
            }
        };

        let player = { x: 400, y: 520, width: 44, height: 36, shield: 100 };
        let lasers = [];
        let enemies = [];
        let particles = [];
        let stars = [];
        let score = 0;
        let isGameRunning = false;
        let keys = {};
        let enemyInterval = null;

        for (let i = 0; i < 80; i++) {
            stars.push({
                x: Math.random() * canvas.width,
                y: Math.random() * canvas.height,
                size: Math.random() * 2,
                speed: (Math.random() * 1.5 + 0.5) * ThemeConfig.starSpeed
            });
        }

        window.addEventListener('keydown', (e) => {
            keys[e.code] = true;
            if (e.code === 'Space' && isGameRunning) {
                fireLaser();
                e.preventDefault();
            }
        });
        window.addEventListener('keyup', (e) => { keys[e.code] = false; });

        function fireLaser() {
            lasers.push({ x: player.x, y: player.y - 15, w: 4, h: 15, speed: 8 });
            SoundEngine.playLaser();
        }

        function spawnEnemyLoop() {
            if (!isGameRunning) return;
            enemies.push({
                x: Math.random() * (canvas.width - 40) + 20,
                y: -30,
                w: 32,
                h: 28,
                speed: Math.random() * 2 + 1.5
            });
            let dynamicDelay = Math.max(400, 1200 - score * 2);
            enemyInterval = setTimeout(spawnEnemyLoop, dynamicDelay);
        }

        function triggerExplosion(x, y, color) {
            for (let i = 0; i < 15; i++) {
                particles.push({
                    x: x, y: y,
                    vx: (Math.random() - 0.5) * 6,
                    vy: (Math.random() - 0.5) * 6,
                    size: Math.random() * 3 + 1,
                    alpha: 1,
                    color: color
                });
            }
            SoundEngine.playExplosion();
        }

        function startSimulation() {
            if (!SoundEngine.ctx) SoundEngine.init();
            document.getElementById('start-screen').classList.add('hidden');
            document.getElementById('gameover-screen').classList.add('hidden');
            resetMetrics();
            isGameRunning = true;
            clearTimeout(enemyInterval);
            spawnEnemyLoop();
        }

        function resetMetrics() {
            score = 0;
            player.shield = 100;
            player.x = 400;
            lasers = [];
            enemies = [];
            particles = [];
            document.getElementById('ui-score').innerText = score;
            document.getElementById('ui-shield').innerText = "100%";
        }

        function frameUpdate() {
            ctx.fillStyle = '#02040a';
            ctx.fillRect(0, 0, canvas.width, canvas.height);

            ctx.fillStyle = 'rgba(255, 255, 255, 0.6)';
            stars.forEach(star => {
                star.y += star.speed;
                if (star.y > canvas.height) star.y = 0;
                ctx.fillRect(star.x, star.y, star.size, star.size);
            });

            if (isGameRunning) {
                if (keys['ArrowLeft'] || keys['KeyA']) player.x = Math.max(30, player.x - 6);
                if (keys['ArrowRight'] || keys['KeyD']) player.x = Math.min(canvas.width - 30, player.x + 6);

                ctx.save();
                ctx.shadowBlur = 15;
                ctx.shadowColor = ThemeConfig.playerColor;
